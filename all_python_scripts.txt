    vlan_url,
    verify=False,
    timeout=10
)

print("VLAN Response:")
print(vlan_response.text)

from credentials import ARUBA2530
import requests
import urllib3

urllib3.disable_warnings()

switch_ip = ARUBA2530["host"]
username = ARUBA2530["username"]
password = ARUBA2530["password"]

session = requests.Session()

# -------------------------
# LOGIN
# -------------------------

login_url = f"https://{switch_ip}/rest/v1/login-sessions"

login_response = session.post(
    login_url,
    json={
        "userName": username,
        "password": password
    },
    verify=False
)

print("Login Status:", login_response.status_code)
print("Login Response:", login_response.text)

# -------------------------
# STORE ARUBA SESSION COOKIE
# -------------------------

login_data = login_response.json()

aruba_cookie = login_data["cookie"]

cookie_name, cookie_value = aruba_cookie.split("=", 1)

session.cookies.set(
    cookie_name,
    cookie_value,
    domain=switch_ip,
    path="/"
)

print("Session cookie stored successfully.")

# -------------------------
# GET VLAN INFORMATION
# -------------------------

vlan_url = f"https://{switch_ip}/rest/v1/vlans"

vlan_response = session.get(
    vlan_url,
    verify=False
)

print("\nVLAN GET Status:", vlan_response.status_code)
print("VLAN Response:")
print(vlan_response.text)

# -------------------------
# LOGOUT
# -------------------------

logout_url = f"https://{switch_ip}/rest/v1/login-sessions"

logout_response = session.delete(
    logout_url,
    verify=False
)

print("\nLogout Status:", logout_response.status_code)
print("Logout Response:", logout_response.text)

from credentials import ARUBA2530
import paramiko
import getpass
import time

switch_ip = ARUBA2530["host"]
ssh_port = ARUBA2530["port"]
username = ARUBA2530["username"]
password = ARUBA2530["password"]

ssh = paramiko.SSHClient()
ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())

ssh.connect(
    hostname=switch_ip,
    port=ssh_port,
    username=username,
    password=password,
    look_for_keys=False,
    allow_agent=False,
    timeout=10
)

print("SSH connection successful!")


shell = ssh.invoke_shell()
time.sleep(2)

# Read initial banner and prompt
while shell.recv_ready():
    print(repr(shell.recv(65535).decode("utf-8", errors="replace")))

# Send the command
shell.sendall(b"\n")
time.sleep(1)

shell.sendall(b"show version\n")
time.sleep(3)

while shell.recv_ready():
    output = shell.recv(65535).decode("utf-8", errors="replace")
    print(output)

ssh.close()
print("SSH connection closed.")

FILE: hp1950_netmiko_dispversion.py
==================================================
from netmiko import ConnectHandler
from credentials import HP1950
import getpass

switch = {
    "device_type": "hp_comware",
    "host": HP1950["host"],
    "port": HP1950["port"],
    "username": HP1950["username"],
    "password": HP1950["password"],
}

connection = ConnectHandler(**switch)

print("\nNetmiko SSH connection successful!")
print("Prompt:", connection.find_prompt())

# Enter extended CLI mode
print("\nEntering extended CLI mode...")

output = connection.send_command_timing("xtd-cli-mode")
print(output)

# Confirm extended CLI mode
output = connection.send_command_timing("Y")
print(output)

# Send extended CLI password
output = connection.send_command_timing(
    HP1950["extended_cli_password"],
    strip_prompt=False,
    strip_command=False
)
print(output)

# Check whether extended CLI authentication succeeded
if "Incorrect password" in output:
    print("Extended CLI authentication failed. Check the password.")
    connection.disconnect()
    raise SystemExit(1)

print("Extended CLI authentication response received.")

# Test command
output = connection.send_command_timing("display version")
print("\n--- DISPLAY VERSION ---")
print(output)

connection.disconnect()
print("\nConnection closed.")
